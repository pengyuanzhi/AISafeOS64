> 是否支持shell功能，是放在用户态还是核心态好 

● 这是一个很好的系统设计问题。让我从多个角度分析：

  Shell功能的价值分析

  Shell的主要用途：
  1. 开发调试：任务管理、内存查看、性能监控
  2. 系统诊断：日志查看、错误分析、状态查询
  3. 生产维护：配置修改、在线调试、故障排查

  用户态 vs 核心态对比

  | 对比项   | 用户态Shell                  | 核心态Shell                |
  |----------|------------------------------|----------------------------|
  | 安全性   | ✅ 隔离性好，bug不影响内核   | ⚠️ 代码bug可能导致内核崩溃 |
  | 功能安全 | ✅ 符合ASIL-D要求            | ⚠️ 增加内核代码复杂度      |
  | 性能     | ⚠️ 需要系统调用开销          | ✅ 直接访问内核数据        |
  | 灵活性   | ✅ 可选编译，动态加载        | ❌ 必须编译进内核          |
  | 调试能力 | ✅ 可通过系统调用扩展        | ✅ 可访问所有内核状态      |
  | 标准兼容 | ✅ 符合POSIX（sh是用户程序） | ❌ 不符合微内核设计        |
  | 代码大小 | ✅ 不增加内核镜像            | ❌ 增加内核镜像大小        |

  推荐方案：混合架构

  ┌─────────────────────────────────────────┐
  │  用户态 Shell (可选)                    │
  │  - 命令解析                              │
  │  - 脚本执行                              │
  │  - 格式化输出                            │
  └──────────────┬──────────────────────────┘
                 │ 系统调用
  ┌──────────────▼──────────────────────────┐
  │  内核调试接口 (核心态，精简)             │
  │  - 安全的查询API                         │
  │  - 受限的修改API                         │
  │  - 性能统计                              │
  └─────────────────────────────────────────┘

  实现策略

  1. 用户态Shell（主要功能）

  // 通过系统调用访问内核信息

      TaskInfo_t tasks[32];
      int count;

      // 系统调用：获取任务列表
      count = syscall_sys_task_info(tasks, 32);

      // 格式化输出（用户态）
      printf("PID   PRI    STATE       TIME\n");
      for (int i = 0; i < count; i++) {
          printf("%-4d  %-3d   %-10s  %llu\n",
                 tasks[i].pid,
                 tasks[i].priority,
                 state_to_str(tasks[i].state),
                 tasks[i].runtime_ns);
      }

      return 0;
  }

  2. 内核调试接口（核心态，最小化）
  // 内核：安全的系统调用实现
  long sys_task_info(TaskInfo_t *tasks, int count) {
      // 参数验证
      if (tasks == NULL) {
          return -EINVAL;
      }

      if (count <= 0 || count > 100) {
          return -EINVAL;
      }

      // 用户空间指针验证
      if (!user_access_ok(tasks, count * sizeof(TaskInfo_t))) {
          return -EFAULT;
      }

      // 复制任务信息到用户空间
      int actual = copy_task_info_to_user(tasks, count);

      return actual;
  }

  // 性能统计（只读）
  long sys_perf_stats(PerfStats_t *stats) {
      if (!user_access_ok(stats, sizeof(PerfStats_t))) {
          return -EFAULT;
      }

      copy_to_user(stats, &g_perf_stats, sizeof(PerfStats_t));

      return 0;
  }

      7103 +#### 22.4.2 mem命令实现                                         
      

      7104 +                                                             
      7105 +```c                                                         
      7106 +/**                                                          
      7107 + * @brief mem命令：显示内存使用统计                             
             

      7108 + * @param args 命令参数（忽略）                                 
          

      7109 + * @return 成功返回0，失败返回负错误码                          
               

      7110 + *                                                           
      7111 + * @note MISRA规则遵守：                                        
       

      7112 + *   - 规则10.1: 检查整数溢出                                   
          

      7113 + *   - 规则10.4: 防止有符号整数溢出                             
             

      7114 + *   - 规则12.3: 避免除零                                       
        

      7115 + *   - 规则13.5: 检查指针参数                                   
          

      7116 + *                                                           
      7117 + * @note 通过系统调用获取内存统计                               
              

      7118 + * @note 计算内存使用百分比                                     
           

      7119 + *                                                           
      7120 + * @warning 必须防止除零错误                                    
          

      7121 + */                                                          
      7122 +static int shell_cmd_mem(const char *args) {                 
      7123 +    MemStats_t stats;                                        
      7124 +    uint32_t used_percent;                                   
      7125 +    uint32_t free_percent;                                   
      7126 +    int ret;                                                 
      7127 +                                                             
      7128 +    /* 抑制未使用参数警告 */                                    
           

      7129 +    (void)args;                                              
      7130 +                                                             
      7131 +    /* 获取内存统计（系统调用） */                              
              

      7132 +    ret = syscall_sys_mem_stats(&stats);                     
      7133 +                                                             
      7134 +    if (ret != 0) {                                          
      7135 +        (void)shell_printf("Error getting memory stats:      
           +%d\n", ret);                                                 
      7136 +        return ret;                                          
      7137 +    }                                                        
      7138 +                                                             
      7139 +    /* 计算百分比（×100避免浮点） */                            
             

      7140 +    if (stats.total > 0U) {                                  
      7141 +        uint64_t used_times_100;                             
      7142 +        uint64_t free_times_100;                             
      7143 +                                                             
      7144 +        /* 计算已使用百分比（×100） */                          
            

      7145 +        used_times_100 = (stats.used * 100ULL) /             
           +stats.total;                                                 
      7146 +        used_percent = (uint32_t)used_times_100;             
      7147 +                                                             
      7148 +        /* 计算空闲百分比（×100） */                            
           

      7149 +        free_times_100 = (stats.free * 100ULL) /             
           +stats.total;                                                 
      7150 +        free_percent = (uint32_t)free_times_100;             
      7151 +    } else {                                                 
      7152 +        used_percent = 0U;                                   
      7153 +        free_percent = 0U;                                   
      7154 +    }                                                        
      7155 +                                                             
      7156 +    /* 打印内存统计 */                                          
        

      7157 +    (void)shell_printf("Total:    %llu KB\n", stats.total /  
           +1024ULL);                                                    
      7158 +    (void)shell_printf("Used:     %llu KB (%u.%02u%%)\n",    
      7159 +                      stats.used / 1024ULL,                  
      7160 +                      used_percent / 100U,                   
      7161 +                      used_percent % 100U);                  
      7162 +    (void)shell_printf("Free:     %llu KB (%u.%02u%%)\n",    
      7163 +                      stats.free / 1024ULL,                  
      7164 +                      free_percent / 100U,                   
      7165 +                      free_percent % 100U);                  
      7166 +    (void)shell_printf("Kernel:   %llu KB\n", stats.kernel / 
           + 1024ULL);                                                   
      7167 +    (void)shell_printf("Tasks:    %llu KB\n", stats.tasks /  
           +1024ULL);                                                    
      7168 +                                                             
      7169 +    return 0;                                                
      7170 +}                                                            
      7171 +```

      7173 +### 22.5 内核调试系统调用MISRA-C编码规范                        
              

      7174 +                                                             
      7175 +#### 22.5.1 任务信息系统调用                                    
          

      7176 +                                                             
      7177 +```c                                                         
      7178 +/**                                                          
      7179 + * @brief 任务信息系统调用实现                                  
            

      7180 + * @param tasks 任务信息数组（用户空间指针）                    
                

      7181 + * @param count 数组大小                                        
      

      7182 + * @return 成功返回实际任务数量，失败返回负错误码               
                     

      7183 + *                                                           
      7184 + * @note MISRA规则遵守：                                        
       

      7185 + *   - 规则10.1: 检查整数溢出                                   
          

      7186 + *   - 规则10.3: 防止整数溢出                                   
          

      7187 + *   - 规则13.5: 检查指针参数                                   
          

      7188 + *   - 规则18.1: 指针运算安全                                   
          

      7189 + *                                                           
      7190 + * @note 仅返回用户有权限查看的任务                             
               

      7191 + * @note 复制到用户空间（user_access_ok验证）                   
             

      7192 + *                                                           
      7193 + * @warning tasks必须指向有效的用户空间内存                     
               

      7194 + * @warning count不能超过最大限制                               
          

      7195 + * @warning 必须验证用户空间指针                                
            

      7196 + */                                                          
      7197 +long sys_task_info(TaskInfo_t *tasks, int count) {           
      7198 +    int actual_count;                                        
      7199 +    int i;                                                   
      7200 +    int ret;                                                 
      7201 +                                                             
      7202 +    /* 参数验证 */                                              
      

      7203 +    if (tasks == NULL) {                                     
      7204 +        return -EINVAL;                                      
      7205 +    }                                                        
      7206 +                                                             
      7207 +    if (count <= 0) {                                        
      7208 +        return -EINVAL;                                      
      7209 +    }                                                        
      7210 +                                                             
      7211 +    if (count > 100) {                                       
      7212 +        /* 限制最大数量 */                                      
        

      7213 +        return -EINVAL;                                      
      7214 +    }                                                        
      7215 +                                                             
      7216 +    /* 验证用户空间指针 */                                      
          

      7217 +    if (!user_access_ok(tasks, (size_t)count *               
           +sizeof(TaskInfo_t))) {                                       
      7218 +        return -EFAULT;                                      
      7219 +    }                                                        
      7220 +                                                             
      7221 +    /* 获取任务信息 */                                          
        

      7222 +    actual_count = get_task_info_internal(tasks, count);     
      7223 +                                                             
      7224 +    if (actual_count < 0) {                                  
      7225 +        return actual_count;                                 
      7226 +    }                                                        
      7227 +                                                             
      7228 +    /* 复制到用户空间 */                                        
         

      7229 +    for (i = 0; i < actual_count; i++) {                     
      7230 +        TaskInfo_t *task = &tasks[i];                        
      7231 +                                                             
      7232 +        /* 检查权限 */                                          
      

      7233 +        if (!has_capability(CAP_SHELL_VIEW)) {               
      7234 +            /* 无权限：跳过敏感信息 */                          
            

      7235 +            task->stack_ptr = 0ULL;                          
      7236 +            task->pc = 0ULL;                                 
      7237 +        }                                                    
      7238 +                                                             
      7239 +        /* 复制到用户空间 */                                    
         

      7240 +        ret = copy_to_user(&tasks[i], task,                  
           +sizeof(TaskInfo_t));                                         
      7241 +                                                             
      7242 +        if (ret != 0) {                                      
      7243 +            return ret;                                      
      7244 +        }                                                    
      7245 +    }                                                        
      7246 +                                                             
      7247 +    return (long)actual_count;                               
      7248 +}                                                            
      7249 +```

● 这是一个非常好的问题！cpio内存文件系统（initramfs）对于嵌入式系统启动非常有价值。让我详细分析一下：

  cpio内存文件系统的价值

  1. 为什么需要cpio内存文件系统？

  传统启动方式的问题：
  Bootloader → 内核 → 挂载根文件系统（NFS/Flash/SD卡）
                             ↓
                      依赖块设备驱动
                      依赖文件系统驱动
                      启动慢，复杂度高

  使用cpio内存文件系统：
  Bootloader → 内核 + cpio镜像 → 直接解压到内存 → 执行init
                                    ↓
                              无需块设备
                              启动速度快
                              简化启动流程

  2. 对AISafe64的优势

  | 优势     | 说明                                   |
  |----------|----------------------------------------|
  | 快速启动 | 文件系统已在内存，无需等待块设备初始化 |
  | 简化部署 | 单一镜像文件（内核+根文件系统）        |
  | 早期配置 | 在块设备驱动加载前即可运行脚本         |
  | 灵活性   | 可以包含配置文件、启动脚本、诊断工具   |
  | 安全性   | 只读根文件系统，防止运行时修改         |
  | 符合RTOS | 嵌入式系统常见做法                     |

  推荐方案：简化的initramfs + rcS脚本

  架构设计

  1. cpio格式支持

  /**
   * @brief cpio文件头（newc格式）
   */
  typedef struct {
      char    magic[6];      /* "070701" or "070702" */
      char    ino[8];        /* inode number */
      char    mode[8];       /* 文件权限和类型 */
      char    uid[8];        /* user ID */
      char    gid[8];        /* group ID */
      char    nlink[8];      /* 链接数量 */
      char    mtime[8];      /* 修改时间 */
      char    filesize[8];   /* 文件大小 */
      char    devmajor[8];   /* 主设备号 */
      char    devminor[8];   /* 次设备号 */
      char    rdevmajor[8];  /* 主设备号（特殊文件） */
      char    rdevminor[8];  /* 次设备号（特殊文件） */
      char    namesize[8];   /* 文件名长度 */
      char    check[8];      /* 校验和 */
  } __attribute__((packed)) CpioHeader_t;

  /**
   * @brief cpio文件系统
   */
  typedef struct {
      uint8_t     *data;          /* cpio镜像数据 */
      uint32_t    size;           /* 镜像大小 */
      bool        mounted;        /* 是否已挂载 */
  } Initramfs_t;

  /**
   * @brief 从cpio镜像中读取文件
   * @param path 文件路径
   * @param buf 输出缓冲区
   * @param size 缓冲区大小
   * @return 成功返回文件大小，失败返回负错误码
   *
   * @note 简化的cpio读取实现
   * @note 仅支持读取（initramfs是只读的）
   */
  int initramfs_read(const char *path, uint8_t *buf, uint32_t size) {
      uint32_t offset = 0;
      CpioHeader_t *hdr;

      /* 遍历cpio归档 */
      while (offset < g_initramfs.size) {
          hdr = (CpioHeader_t *)(g_initramfs.data + offset);

          /* 解析文件头 */
          uint32_t namesize = cpio_parse_hex(hdr->namesize);
          uint32_t filesize = cpio_parse_hex(hdr->filesize);
          char *name = (char *)(hdr + 1);

          /* 检查是否是结束标记 */
          if (strcmp(name, "TRAILER!!!") == 0) {
              break;
          }

          /* 检查文件名是否匹配 */
          if (strcmp(path, name) == 0) {
              /* 找到文件，复制内容 */
              uint32_t copy_size = (filesize < size) ? filesize : size;
              uint8_t *filedata = (uint8_t *)name + namesize;

              /* 对齐到4字节边界 */
              filedata += (4 - (namesize % 4)) % 4;

              memcpy(buf, filedata, copy_size);
              return (int)copy_size;
          }

          /* 跳到下一个文件 */
          offset += sizeof(CpioHeader_t) + namesize + filesize;
          offset = (offset + 3) & ~3U;  /* 4字节对齐 */
      }

      return -ENOENT;
  }

  rcS脚本解释器实现：

  /**
   * @brief rcS脚本命令
   */
  typedef struct {
      const char *name;
      int (*handler)(int argc, char *argv[]);
  } RcScriptCmd_t;

  /**
   * @brief 简化的rcS脚本解释器
   * @param script_path 脚本路径
   * @return 成功返回0，失败返回负错误码
   *
   * @note 仅支持简化的Shell语法
   * @note 不支持管道、重定向、后台运行
   */
  int rc_script_execute(const char *script_path) {
      char *script;
      uint32_t script_size;
      char line[256];
      uint32_t line_pos = 0;
      int ret;

      /* 读取脚本文件 */
      script_size = initramfs_read(script_path, (uint8_t *)g_script_buf,
                                    sizeof(g_script_buf));
      if (script_size < 0) {
          return script_size;
      }

      script = g_script_buf;

      /* 逐行解析和执行 */
      while (line_pos < script_size) {
          char *line_start = &script[line_pos];
          char *newline;
          uint32_t line_len;

          /* 查找换行符 */
          newline = strchr(line_start, '\n');
          if (newline == NULL) {
              break;
          }

          line_len = (uint32_t)(newline - line_start);

          /* 复制行到缓冲区 */
          if (line_len >= sizeof(line)) {
              line_len = sizeof(line) - 1U;
          }
          memcpy(line, line_start, line_len);
          line[line_len] = '\0';

          /* 跳过空行和注释 */
          if (line[0] == '\0' || line[0] == '#') {
              line_pos += line_len + 1;
              continue;
          }

          /* 解析和执行命令 */
          ret = rc_execute_line(line);
          if (ret != 0) {
              printf("rcS: Error executing line: %s\n", line);
              return ret;
          }

          line_pos += line_len + 1;
      }

      return 0;
  }

  /**
   * @brief 执行rcS脚本的一行
   */
  static int rc_execute_line(const char *line) {
      char *argv[16];
      int argc;
      const RcScriptCmd_t *cmd;
      char *line_copy;
      int ret;

      /* 复制行（用于修改） */
      line_copy = strdup(line);
      if (line_copy == NULL) {
          return -ENOMEM;
      }

      /* 解析命令和参数 */
      argc = rc_parse_args(line_copy, argv, 16);
      if (argc <= 0) {
          free(line_copy);
          return 0;
      }

      /* 查找命令 */
      cmd = rc_find_command(argv[0]);
      if (cmd == NULL) {
          printf("rcS: Unknown command: %s\n", argv[0]);
          free(line_copy);
          return -ENOENT;
      }

      /* 执行命令 */
      ret = cmd->handler(argc, argv);

      free(line_copy);
      return ret;
  }

  /**
   * @brief rcS命令表
   */
  static const RcScriptCmd_t g_rc_commands[] = {
      /* 文件系统命令 */
      { "mount",    rc_cmd_mount },
      { "umount",   rc_cmd_umount },

      /* 网络命令 */

  # 构建initramfs
  initramfs: $(shell find initramfs -type f)
  	@echo "Building initramfs..."
  	@cd initramfs && find . | cpio -o -H newc > ../initramfs.cpio
  	@gzip -f initramfs.cpio
  	@ls -lh initramfs.cpio.gz

  # 将initramfs打包到内核镜像
  kernel: initramfs
  	@echo "Linking kernel with initramfs..."
  	$(LD) -T kernel.lds \
  	    -o vmlinux \
  	    --oformat=elf64-littleaarch64 \
  	    --whole-archive \
  	    kernel.o \
  	    initramfs.o \
  	    -Map kernel.map

  # initramfs.o从二进制文件生成
  initramfs.o: initramfs.cpio.gz
  	$(OBJCOPY) -I binary -O elf64-littleaarch64 \
  	    -B aarch64 \
  	    --rename-section .data=.initramfs \
  	    initramfs.cpio.gz initramfs.o

  /**

  int safe_initramfs_read(const char *path, uint8_t *buf, uint32_t size) {

       889 +/**

       902 + * @param buf 输出缓冲区                                        
       

       903 + * @param size 缓冲区大小                                       
       

       904 + * @return 成功返回文件大小，失败返回负错误码                   
                   

       905 + *                                                           
       906 + * @note initramfs是只读的                                      
      

       907 + * @note 不支持写操作                                           
        

       908 + * @note 支持符号链接（简化的实现）                             
               

       909 + */                                                          
       910 +int initramfs_read(const char *path, uint8_t *buf, uint32_t  
           +size) {                                                      
       911 +    uint32_t offset = 0;                                     
       912 +                                                             
       913 +    /* 参数验证 */                                              
      

       914 +    if (path == NULL || buf == NULL) {                       
       915 +        return -EINVAL;                                      
       916 +    }                                                        
       917 +                                                             
       918 +    if (size == 0U) {                                        
       919 +        return -EINVAL;                                      
       920 +    }                                                        
       921 +                                                             
       922 +    /* 遍历cpio归档 */                                          
      

       923 +    while (offset < g_initramfs.size) {                      
       924 +        CpioHeader_t *hdr;                                   
       925 +        uint32_t namesize;                                   
       926 +        uint32_t filesize;                                   
       927 +        char *name;                                          
       928 +                                                             
       929 +        hdr = (CpioHeader_t *)(g_initramfs.data + offset);   
       930 +                                                             
       931 +        /* 验证魔数 */                                          
      

       932 +        if (memcmp(hdr->magic, "070701", 6) != 0 &&          
       933 +            memcmp(hdr->magic, "070702", 6) != 0) {          
       934 +            return -EINVAL;                                  
       935 +        }                                                    
       936 +                                                             
       937 +        /* 解析文件头 */                                        
       

       938 +        namesize = cpio_parse_hex(hdr->namesize);            
       939 +        filesize = cpio_parse_hex(hdr->filesize);            
       940 +        name = (char *)(hdr + 1);                            
       941 +                                                             
       942 +        /* 检查是否是结束标记 */                                
           

       943 +        if (strcmp(name, "TRAILER!!!") == 0) {               
       944 +            break;                                           
       945 +        }                                                    
       946 +                                                             
       947 +        /* 检查文件名是否匹配 */                                
           

       948 +        if (strcmp(path, name) == 0) {                       
       949 +            /* 找到文件，复制内容 */                            
           

       950 +            uint32_t copy_size = (filesize < size) ?         
           +filesize : size;                                             
       951 +            uint8_t *filedata = (uint8_t *)name + namesize;  
       952 +                                                             
       953 +            /* 对齐到4字节边界 */                               
         

       954 +            filedata += (4U - (namesize & 3U)) & 3U;         
       955 +                                                             
       956 +            (void)memcpy(buf, filedata, copy_size);          
       957 +            return (int)copy_size;                           
       958 +        }                                                    
       959 +                                                             
       960 +        /* 跳到下一个文件 */                                    
         

       961 +        offset += sizeof(CpioHeader_t) + namesize +          
           +filesize;                                                    
       962 +        offset = (offset + 3U) & ~3U;  /* 4字节对齐 */          
      

       963 +    }                                                        
       964 +                                                             
       965 +    return -ENOENT;                                          
       966 +}                                                            
       967 +                                                             
       968 +/**                                                          
       969 + * @brief 解析ASCII十六进制字符串                               
           

       970 + * @param str ASCII十六进制字符串                               
         

       971 + * @return 解析后的数值                                         
        

       972 + */                                                          
       973 +static uint32_t cpio_parse_hex(const char *str) {            
       974 +    uint32_t val = 0U;                                       
       975 +                                                             
       976 +    while (*str != '\0') {                                   
       977 +        uint8_t ch = (uint8_t)*str;                          
       978 +                                                             
       979 +        if (ch >= '0' && ch <= '9') {                        
       980 +            val = (val << 4U) + (uint32_t)(ch - '0');        
       981 +        } else if (ch >= 'a' && ch <= 'f') {                 
       982 +            val = (val << 4U) + (uint32_t)(ch - 'a' + 10U);  
       983 +        } else if (ch >= 'A' && ch <= 'F') {                 
       984 +            val = (val << 4U) + (uint32_t)(ch - 'A' + 10U);  
       985 +        } else {                                             
       986 +            break;                                           
       987 +        }                                                    
       988 +                                                             
       989 +        str++;                                               
       990 +    }                                                        
       991 +                                                             
       992 +    return val;                                              
       993 +}

      1042 +**rcS脚本解释器实现**：                                         
          

      1043 +                                                             
      1044 +```c                                                         
      1045 +/**                                                          
      1046 + * @brief rcS脚本命令处理函数                                   
          

      1047 + */                                                          
      1048 +typedef int (*RcCmdHandler_t)(int argc, char *argv[]);       
      1049 +                                                             
      1050 +/**                                                          
      1051 + * @brief rcS命令表项                                           
      

      1052 + */                                                          
      1053 +typedef struct {

      1055 +    RcCmdHandler_t   handler;     /* 处理函数 */                
      

      1056 +    const char       *help;       /* 帮助信息 */                
      

      1057 +} RcCommand_t;                                               
      1058 +                                                             
      1059 +/**                                                          
      1060 + * @brief 执行rcS脚本                                           
      

      1061 + * @param script_path 脚本路径（如"/etc/rcS"）                  
         

      1062 + * @return 成功返回0，失败返回负错误码                          
               

      1063 + *                                                           
      1064 + * @note 仅支持简化的Shell语法                                  
          

      1065 + * @note 不支持管道、重定向、后台运行                           
                

      1066 + * @note 支持注释（#开头）                                      
          

      1067 + */                                                          
      1068 +int rc_script_execute(const char *script_path) {             
      1069 +    char *script;                                            
      1070 +    uint32_t script_size;                                    
      1071 +    char line[256];                                          
      1072 +    uint32_t line_pos = 0;                                   
      1073 +    int ret;                                                 
      1074 +                                                             
      1075 +    /* 读取脚本文件 */                                          
        

      1076 +    script_size = initramfs_read(script_path, (uint8_t       
           +*)g_rc_script_buf,                                           
      1077 +                                  sizeof(g_rc_script_buf));  
      1078 +    if (script_size < 0) {                                   
      1079 +        printf("rcS: Failed to read script: %s\n",           
           +script_path);                                                
      1080 +        return script_size;                                  
      1081 +    }                                                        
      1082 +                                                             
      1083 +    script = g_rc_script_buf;                                
      1084 +                                                             
      1085 +    /* 逐行解析和执行 */                                        
         

      1086 +    while (line_pos < script_size) {                         
      1087 +        char *line_start = &script[line_pos];                
      1088 +        char *newline;                                       
      1089 +        uint32_t line_len;                                   
      1090 +                                                             
      1091 +        /* 查找换行符 */                                        
       

      1092 +        newline = strchr(line_start, '\n');                  
      1093 +        if (newline == NULL) {                               
      1094 +            break;                                           
      1095 +        }                                                    
      1096 +                                                             
      1097 +        line_len = (uint32_t)(newline - line_start);         
      1098 +                                                             
      1099 +        /* 跳过空行 */                                          
      

      1100 +        if (line_len == 0U) {                                
      1101 +            line_pos += 1U;                                  
      1102 +            continue;                                        
      1103 +        }                                                    
      1104 +                                                             
      1105 +        /* 复制行到缓冲区 */                                    
         

      1106 +        if (line_len >= sizeof(line)) {                      
      1107 +            line_len = sizeof(line) - 1U;                    
      1108 +        }                                                    
      1109 +        (void)memcpy(line, line_start, line_len);            
      1110 +        line[line_len] = '\0';                               
      1111 +                                                             
      1112 +        /* 跳过注释 */                                          
      

      1113 +        if (line[0] == '#') {                                
      1114 +            line_pos += line_len + 1U;                       
      1115 +            continue;                                        
      1116 +        }                                                    
      1117 +                                                             
      1118 +        /* 执行命令 */                                          
      

      1119 +        ret = rc_execute_line(line);                         
      1120 +        if (ret != 0) {                                      
      1121 +            printf("rcS: Error executing line %u: %s (error: 
           + %d)\n",                                                     
      1122 +                   line_pos, line, ret);                     
      1123 +            /* 继续执行，不中断启动 */                          
            

      1124 +        }                                                    
      1125 +                                                             
      1126 +        line_pos += line_len + 1U;                           
      1127 +    }                                                        
      1128 +                                                             
      1129 +    return 0;                                                
      1130 +}                                                            
      1131 +                                                             
      1132 +/**                                                          
      1133 + * @brief 执行rcS脚本的一行                                     
         

      1134 + */

      1136 +    char *argv[16];

      1138 +    const RcCommand_t *cmd;                                  
      1139 +    int ret;                                                 
      1140 +                                                             
      1141 +    /* 解析命令和参数 */                                        
         

      1142 +    argc = rc_parse_args(line, argv, 16);                    
      1143 +    if (argc <= 0) {                                         
      1144 +        return 0;                                            
      1145 +    }                                                        
      1146 +                                                             
      1147 +    /* 查找命令 */                                              
      

      1148 +    cmd = rc_find_command(argv[0]);                          
      1149 +    if (cmd == NULL) {                                       
      1150 +        printf("rcS: Unknown command: %s\n", argv[0]);       
      1151 +        return -ENOENT;

      1262 +**配置文件解析实现**：                                          
           

      1263 +                                                             
      1264 +```c                                                         
      1265 +/**                                                          
      1266 + * @brief INI配置文件解析                                       
        

      1267 + * @param path 配置文件路径                                     
        

      1268 + * @param callback 配置项回调函数                               
         

      1269 + * @param context 回调上下文                                    
       

      1270 + * @return 成功返回0，失败返回负错误码                          
               

      1271 + */                                                          
      1272 +typedef int (*IniCallback_t)(const char *section,            
      1273 +                             const char *key,                
      1274 +                             const char *value,              
      1275 +                             void *context);                 
      1276 +                                                             
      1277 +int ini_parse(const char *path, IniCallback_t callback, void 
           + *context) {                                                 
      1278 +    char *data;                                              
      1279 +    uint32_t size;                                           
      1280 +    char line[256];                                          
      1281 +    uint32_t pos = 0;                                        
      1282 +    char current_section[64] = {0};                          
      1283 +                                                             
      1284 +    /* 读取配置文件 */                                          
        

      1285 +    size = initramfs_read(path, (uint8_t *)g_ini_buf,        
           +sizeof(g_ini_buf));                                          
      1286 +    if (size < 0) {                                          
      1287 +        return size;                                         
      1288 +    }                                                        
      1289 +                                                             
      1290 +    data = g_ini_buf;                                        
      1291 +                                                             
      1292 +    /* 逐行解析 */                                              
      

      1293 +    while (pos < size) {                                     
      1294 +        char *line_start = &data[pos];                       
      1295 +        char *newline;                                       
      1296 +        uint32_t line_len;                                   
      1297 +                                                             
      1298 +        newline = strchr(line_start, '\n');                  
      1299 +        if (newline == NULL) {                               
      1300 +            break;                                           
      1301 +        }                                                    
      1302 +                                                             
      1303 +        line_len = (uint32_t)(newline - line_start);         
      1304 +        if (line_len >= sizeof(line)) {                      
      1305 +            line_len = sizeof(line) - 1U;                    
      1306 +        }                                                    
      1307 +                                                             
      1308 +        (void)memcpy(line, line_start, line_len);            
      1309 +        line[line_len] = '\0';                               
      1310 +                                                             
      1311 +        /* 跳过空行和注释 */                                    
         

      1312 +        if (line[0] == '\0' || line[0] == '#' || line[0] ==  
           +';') {                                                       
      1313 +            pos += line_len + 1U;                            
      1314 +            continue;                                        
      1315 +        }                                                    
      1316 +                                                             
      1317 +        /* 解析section */                                      
      1318 +        if (line[0] == '[') {                                
      1319 +            char *end = strchr(line, ']');                   
      1320 +            if (end != NULL) {                               
      1321 +                uint32_t len = (uint32_t)(end - line - 1U);  
      1322 +                if (len < sizeof(current_section)) {         
      1323 +                    (void)memcpy(current_section, line + 1,  
           +len);                                                        
      1324 +                    current_section[len] = '\0';             
      1325 +                }                                            
      1326 +            }                                                
      1327 +        }                                                    
      1328 +        /* 解析key=value */                                    
      1329 +        else {                                               
      1330 +            char *eq = strchr(line, '=');                    
      1331 +            if (eq != NULL) {                                
      1332 +                *eq = '\0';                                  
      1333 +                char *key = line;                            
      1334 +                char *value = eq + 1;                        
      1335 +                                                             
      1336 +                /* 去除空格 */                                  
      

      1337 +                char *key_end = eq - 1;                      
      1338 +                while (key_end > key && *key_end == ' ') {   
      1339 +                    *key_end = '\0';                         
      1340 +                    key_end--;                               
      1341 +                }                                            
      1342 +                                                             
      1343 +                while (*value == ' ') {                      
      1344 +                    value++;                                 
      1345 +                }                                            
      1346 +                                                             
      1347 +                /* 调用回调 */                                  
      

      1348 +                if (callback != NULL) {                      
      1349 +                    (void)callback(current_section, key,     
           +value, context);                                             
      1350 +                }                                            
      1351 +            }                                                
      1352 +        }                                                    
      1353 +                                                             
      1354 +        pos += line_len + 1U;                                
      1355 +    }                                                        
      1356 +                                                             
      1357 +    return 0;                                                
      1358 +}                                                            
      1359 +```
