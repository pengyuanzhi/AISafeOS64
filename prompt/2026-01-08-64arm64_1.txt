> 是否支持shell功能，是放在用户态还是核心态好 

● 这是一个很好的系统设计问题。让我从多个角度分析：

  Shell功能的价值分析

  Shell的主要用途：
  1. 开发调试：任务管理、内存查看、性能监控
  2. 系统诊断：日志查看、错误分析、状态查询
  3. 生产维护：配置修改、在线调试、故障排查

  用户态 vs 核心态对比

  | 对比项   | 用户态Shell                  | 核心态Shell                |
  |----------|------------------------------|----------------------------|
  | 安全性   | ✅ 隔离性好，bug不影响内核   | ⚠️ 代码bug可能导致内核崩溃 |
  | 功能安全 | ✅ 符合ASIL-D要求            | ⚠️ 增加内核代码复杂度      |
  | 性能     | ⚠️ 需要系统调用开销          | ✅ 直接访问内核数据        |
  | 灵活性   | ✅ 可选编译，动态加载        | ❌ 必须编译进内核          |
  | 调试能力 | ✅ 可通过系统调用扩展        | ✅ 可访问所有内核状态      |
  | 标准兼容 | ✅ 符合POSIX（sh是用户程序） | ❌ 不符合微内核设计        |
  | 代码大小 | ✅ 不增加内核镜像            | ❌ 增加内核镜像大小        |

  推荐方案：混合架构

  ┌─────────────────────────────────────────┐
  │  用户态 Shell (可选)                    │
  │  - 命令解析                              │
  │  - 脚本执行                              │
  │  - 格式化输出                            │
  └──────────────┬──────────────────────────┘
                 │ 系统调用
  ┌──────────────▼──────────────────────────┐
  │  内核调试接口 (核心态，精简)             │
  │  - 安全的查询API                         │
  │  - 受限的修改API                         │
  │  - 性能统计                              │
  └─────────────────────────────────────────┘

  实现策略

  1. 用户态Shell（主要功能）

  // 通过系统调用访问内核信息

      TaskInfo_t tasks[32];
      int count;

      // 系统调用：获取任务列表
      count = syscall_sys_task_info(tasks, 32);

      // 格式化输出（用户态）
      printf("PID   PRI    STATE       TIME\n");
      for (int i = 0; i < count; i++) {
          printf("%-4d  %-3d   %-10s  %llu\n",
                 tasks[i].pid,
                 tasks[i].priority,
                 state_to_str(tasks[i].state),
                 tasks[i].runtime_ns);
      }

      return 0;
  }

  2. 内核调试接口（核心态，最小化）
  // 内核：安全的系统调用实现
  long sys_task_info(TaskInfo_t *tasks, int count) {
      // 参数验证
      if (tasks == NULL) {
          return -EINVAL;
      }

      if (count <= 0 || count > 100) {
          return -EINVAL;
      }

      // 用户空间指针验证
      if (!user_access_ok(tasks, count * sizeof(TaskInfo_t))) {
          return -EFAULT;
      }

      // 复制任务信息到用户空间
      int actual = copy_task_info_to_user(tasks, count);

      return actual;
  }

  // 性能统计（只读）
  long sys_perf_stats(PerfStats_t *stats) {
      if (!user_access_ok(stats, sizeof(PerfStats_t))) {
          return -EFAULT;
      }

      copy_to_user(stats, &g_perf_stats, sizeof(PerfStats_t));

      return 0;
  }

      7103 +#### 22.4.2 mem命令实现                                         
      

      7104 +                                                             
      7105 +```c                                                         
      7106 +/**                                                          
      7107 + * @brief mem命令：显示内存使用统计                             
             

      7108 + * @param args 命令参数（忽略）                                 
          

      7109 + * @return 成功返回0，失败返回负错误码                          
               

      7110 + *                                                           
      7111 + * @note MISRA规则遵守：                                        
       

      7112 + *   - 规则10.1: 检查整数溢出                                   
          

      7113 + *   - 规则10.4: 防止有符号整数溢出                             
             

      7114 + *   - 规则12.3: 避免除零                                       
        

      7115 + *   - 规则13.5: 检查指针参数                                   
          

      7116 + *                                                           
      7117 + * @note 通过系统调用获取内存统计                               
              

      7118 + * @note 计算内存使用百分比                                     
           

      7119 + *                                                           
      7120 + * @warning 必须防止除零错误                                    
          

      7121 + */                                                          
      7122 +static int shell_cmd_mem(const char *args) {                 
      7123 +    MemStats_t stats;                                        
      7124 +    uint32_t used_percent;                                   
      7125 +    uint32_t free_percent;                                   
      7126 +    int ret;                                                 
      7127 +                                                             
      7128 +    /* 抑制未使用参数警告 */                                    
           

      7129 +    (void)args;                                              
      7130 +                                                             
      7131 +    /* 获取内存统计（系统调用） */                              
              

      7132 +    ret = syscall_sys_mem_stats(&stats);                     
      7133 +                                                             
      7134 +    if (ret != 0) {                                          
      7135 +        (void)shell_printf("Error getting memory stats:      
           +%d\n", ret);                                                 
      7136 +        return ret;                                          
      7137 +    }                                                        
      7138 +                                                             
      7139 +    /* 计算百分比（×100避免浮点） */                            
             

      7140 +    if (stats.total > 0U) {                                  
      7141 +        uint64_t used_times_100;                             
      7142 +        uint64_t free_times_100;                             
      7143 +                                                             
      7144 +        /* 计算已使用百分比（×100） */                          
            

      7145 +        used_times_100 = (stats.used * 100ULL) /             
           +stats.total;                                                 
      7146 +        used_percent = (uint32_t)used_times_100;             
      7147 +                                                             
      7148 +        /* 计算空闲百分比（×100） */                            
           

      7149 +        free_times_100 = (stats.free * 100ULL) /             
           +stats.total;                                                 
      7150 +        free_percent = (uint32_t)free_times_100;             
      7151 +    } else {                                                 
      7152 +        used_percent = 0U;                                   
      7153 +        free_percent = 0U;                                   
      7154 +    }                                                        
      7155 +                                                             
      7156 +    /* 打印内存统计 */                                          
        

      7157 +    (void)shell_printf("Total:    %llu KB\n", stats.total /  
           +1024ULL);                                                    
      7158 +    (void)shell_printf("Used:     %llu KB (%u.%02u%%)\n",    
      7159 +                      stats.used / 1024ULL,                  
      7160 +                      used_percent / 100U,                   
      7161 +                      used_percent % 100U);                  
      7162 +    (void)shell_printf("Free:     %llu KB (%u.%02u%%)\n",    
      7163 +                      stats.free / 1024ULL,                  
      7164 +                      free_percent / 100U,                   
      7165 +                      free_percent % 100U);                  
      7166 +    (void)shell_printf("Kernel:   %llu KB\n", stats.kernel / 
           + 1024ULL);                                                   
      7167 +    (void)shell_printf("Tasks:    %llu KB\n", stats.tasks /  
           +1024ULL);                                                    
      7168 +                                                             
      7169 +    return 0;                                                
      7170 +}                                                            
      7171 +```

      7173 +### 22.5 内核调试系统调用MISRA-C编码规范                        
              

      7174 +                                                             
      7175 +#### 22.5.1 任务信息系统调用                                    
          

      7176 +                                                             
      7177 +```c                                                         
      7178 +/**                                                          
      7179 + * @brief 任务信息系统调用实现                                  
            

      7180 + * @param tasks 任务信息数组（用户空间指针）                    
                

      7181 + * @param count 数组大小                                        
      

      7182 + * @return 成功返回实际任务数量，失败返回负错误码               
                     

      7183 + *                                                           
      7184 + * @note MISRA规则遵守：                                        
       

      7185 + *   - 规则10.1: 检查整数溢出                                   
          

      7186 + *   - 规则10.3: 防止整数溢出                                   
          

      7187 + *   - 规则13.5: 检查指针参数                                   
          

      7188 + *   - 规则18.1: 指针运算安全                                   
          

      7189 + *                                                           
      7190 + * @note 仅返回用户有权限查看的任务                             
               

      7191 + * @note 复制到用户空间（user_access_ok验证）                   
             

      7192 + *                                                           
      7193 + * @warning tasks必须指向有效的用户空间内存                     
               

      7194 + * @warning count不能超过最大限制                               
          

      7195 + * @warning 必须验证用户空间指针                                
            

      7196 + */                                                          
      7197 +long sys_task_info(TaskInfo_t *tasks, int count) {           
      7198 +    int actual_count;                                        
      7199 +    int i;                                                   
      7200 +    int ret;                                                 
      7201 +                                                             
      7202 +    /* 参数验证 */                                              
      

      7203 +    if (tasks == NULL) {                                     
      7204 +        return -EINVAL;                                      
      7205 +    }                                                        
      7206 +                                                             
      7207 +    if (count <= 0) {                                        
      7208 +        return -EINVAL;                                      
      7209 +    }                                                        
      7210 +                                                             
      7211 +    if (count > 100) {                                       
      7212 +        /* 限制最大数量 */                                      
        

      7213 +        return -EINVAL;                                      
      7214 +    }                                                        
      7215 +                                                             
      7216 +    /* 验证用户空间指针 */                                      
          

      7217 +    if (!user_access_ok(tasks, (size_t)count *               
           +sizeof(TaskInfo_t))) {                                       
      7218 +        return -EFAULT;                                      
      7219 +    }                                                        
      7220 +                                                             
      7221 +    /* 获取任务信息 */                                          
        

      7222 +    actual_count = get_task_info_internal(tasks, count);     
      7223 +                                                             
      7224 +    if (actual_count < 0) {                                  
      7225 +        return actual_count;                                 
      7226 +    }                                                        
      7227 +                                                             
      7228 +    /* 复制到用户空间 */                                        
         

      7229 +    for (i = 0; i < actual_count; i++) {                     
      7230 +        TaskInfo_t *task = &tasks[i];                        
      7231 +                                                             
      7232 +        /* 检查权限 */                                          
      

      7233 +        if (!has_capability(CAP_SHELL_VIEW)) {               
      7234 +            /* 无权限：跳过敏感信息 */                          
            

      7235 +            task->stack_ptr = 0ULL;                          
      7236 +            task->pc = 0ULL;                                 
      7237 +        }                                                    
      7238 +                                                             
      7239 +        /* 复制到用户空间 */                                    
         

      7240 +        ret = copy_to_user(&tasks[i], task,                  
           +sizeof(TaskInfo_t));                                         
      7241 +                                                             
      7242 +        if (ret != 0) {                                      
      7243 +            return ret;                                      
      7244 +        }                                                    
      7245 +    }                                                        
      7246 +                                                             
      7247 +    return (long)actual_count;                               
      7248 +}                                                            
      7249 +```
