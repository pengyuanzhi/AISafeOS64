      10591 +    /* 初始化所有CPU的运行队列 */                              
            

      10592 +    for (cpu = 0U; cpu < MAX_CPUS; cpu++) {                 
      10593 +        ret = rq_init(&per_cpu_rq[cpu], cpu);               
      10594 +        if (ret != 0) {                                     
      10595 +            /* 清理已初始化的运行队列 */                       
             

      10596 +            cleanup_rq(cpu);                                
      10597 +            return ret;                                     
      10598 +        }                                                   
      10599 +    }                                                       
      10600 +                                                            
      10601 +    /* 初始化调度类链表 */                                     
          

      10602 +    sched_class_h = NULL;                                   
      10603 +    register_sched_class(&sched_class_fifo);                
      10604 +    register_sched_class(&sched_class_edf);                 
      10605 +    register_sched_class(&sched_class_cfs);                 
      10606 +    register_sched_class(&sched_class_rr);                  
      10607 +    register_sched_class(&sched_class_idle);                
      10608 +                                                            
      10609 +    scheduler_initialized = 1U;                             
      10610 +    return 0;                                               
      10611 +}                                                           
      10612 +```

      10614 +#### 26.2.2 选择下一个任务                                     
         

      10615 +                                                            
      10616 +```c                                                        
      10617 +/**                                                         
      10618 + * @brief 选择下一个运行的任务                                 
            

      10619 + * @param rq 运行队列指针                                      
        

      10620 + * @return 任务控制块指针，无任务返回idle任务                  
                 

      10621 + *                                                          
      10622 + * @details 核心调度函数，遍历调度类链表                       
                

      10623 + *                                                          
      10624 + * @note MISRA规则遵守：                                       
       

      10625 + *   - 规则13.5: 检查指针参数                                  
          

      10626 + *   - 规则18.1: 指针运算不超过数组范围                        
               

      10627 + *   - 规则1.3: signedness一致性                               
       

      10628 + *   - 规则15.5: 避免多余的循环                                
           

      10629 + *                                                          
      10630 + * @note 时间复杂度: O(M) + O(1) = O(M)                        
       

      10631 + *       - M: 调度类数量（5）                                  
         

      10632 + *       - pick_next: O(1)                                  
      10633 + * @note 总开销: ~185ns                                        

      10634 + *                                                          
      10635 + * @warning rq必须指向有效的运行队列                           
             

      10636 + * @warning 必须持有rq锁                                       
       

      10637 + * @warning 返回值不能为NULL                                   
        

      10638 + */                                                         
      10639 +TCB_t *pick_next_task(struct rq *rq) {                      
      10640 +    const SchedClass_t *class;                              
      10641 +    TCB_t *next_task;                                       
      10642 +                                                            
      10643 +    /* 参数验证 */                                             
      

      10644 +    if (rq == NULL) {                                       
      10645 +        return NULL;                                        
      10646 +    }                                                       
      10647 +                                                            
      10648 +    /* 遍历调度类链表 */                                       
         

      10649 +    for (class = sched_class_h; class != NULL; class =      
            +class->next) {                                              
      10650 +        /* 检查是否有可运行任务 */                             
            

      10651 +        if (rq->nr_running[class->id] == 0U) {              
      10652 +            continue;                                       
      10653 +        }                                                   
      10654 +                                                            
      10655 +        /* 调用调度类的pick_next函数 */                        
          

      10656 +        next_task = class->pick_next(rq);                   
      10657 +                                                            
      10658 +        /* 检查返回值 */                                       
       

      10659 +        if (next_task != NULL) {                            
      10660 +            return next_task;                               
      10661 +        }                                                   
      10662 +    }                                                       
      10663 +                                                            
      10664 +    /* 所有调度类都没有任务，返回idle任务 */                   
                 

      10665 +    return rq->idle;                                        
      10666 +}                                                           
      10667 +```

      10694 +                                                            
      10695 +    /* 参数验证 */

      10709 +    }

      10714 +    }                                                       
      10715 +

      10725 +### 26.3 FIFO调度类实现MISRA-C编码规范                         
           

      10726 +                                                            
      10727 +#### 26.3.1 FIFO初始化                                         

      10728 +                                                            
      10729 +```c                                                        
      10730 +/**                                                         
      10731 + * @brief FIFO调度类初始化                                     
        

      10732 + * @param rq 运行队列指针                                      
        

      10733 + * @return 成功返回0                                           
      

      10734 + *                                                          
      10735 + * @details 初始化256级优先级队列                              
           

      10736 + *                                                          
      10737 + * @note MISRA规则遵守：                                       
       

      10738 + *   - 规则13.5: 检查指针参数                                  
          

      10739 + *   - 规则21.3: 数组索引边界检查                              
            

      10740 + *   - 规则17.3: 指针算术不超过对象大小                        
               

      10741 + *                                                          
      10742 + * @note 分配FIFO私有数据                                      
        

      10743 + * @note 初始化优先级位图                                      
          

      10744 + * @note 时间复杂度: O(256) = O(1)                             
       

      10745 + */                                                         
      10746 +static int fifo_sched_init(struct rq *rq) {                 
      10747 +    struct fifo_rq *fifo_rq;                                
      10748 +    uint32_t i;                                             
      10749 +                                                            
      10750 +    /* 参数验证 */                                             
      

      10751 +    if (rq == NULL) {                                       
      10752 +        return -EINVAL;                                     
      10753 +    }                                                       
      10754 +                                                            
      10755 +    /* 分配FIFO私有数据 */                                     
        

      10756 +    fifo_rq = (struct fifo_rq *)malloc(sizeof(struct        
            +fifo_rq));                                                  
      10757 +    if (fifo_rq == NULL) {                                  
      10758 +        return -ENOMEM;                                     
      10759 +    }                                                       
      10760 +                                                            
      10761 +    /* 初始化优先级位图 */                                     
          

      10762 +    (void)memset(fifo_rq->priority_bitmap, 0,               
            +sizeof(fifo_rq->priority_bitmap));                          
      10763 +                                                            
      10764 +    /* 初始化每个优先级的队列 */                               
             

      10765 +    for (i = 0U; i < 256U; i++) {                           
      10766 +        INIT_LIST_HEAD(&fifo_rq->queue_array[i]);           
      10767 +    }                                                       
      10768 +                                                            
      10769 +    /* 关联到运行队列 */                                       
         

      10770 +    rq->fifo_rq = fifo_rq;                                  
      10771 +                                                            
      10772 +    return 0;                                               
      10773 +}                                                           
      10774 +```                                                         
      10775 +                                                            
      10776 +#### 26.3.2 FIFO入队                                          
      10777 +                                                            
      10778 +```c                                                        
      10779 +/**                                                         
      10780 + * @brief FIFO任务入队                                         
      

      10781 + * @param rq 运行队列指针                                      
        

      10782 + * @param task 任务控制块指针                                  
         

      10783 + *                                                          
      10784 + * @details 将任务加入对应优先级队列尾部                       
                

      10785 + *                                                          
      10786 + * @note MISRA规则遵守：                                       
       

      10787 + *   - 规则13.5: 检查指针参数                                  
          

      10788 + *   - 规则21.3: 数组索引范围                                  
          

      10789 + *   - 规则18.1: 指针运算边界                                  
          

      10790 + *   - 规则17.3: 类型匹配                                      
        

      10791 + *                                                          
      10792 + * @note 时间复杂度: O(1)                                      
       

      10793 + * @note 更新优先级位图                                        
         

      10794 + * @note 同优先级FIFO                                          
      

      10795 + *                                                          
      10796 + * @warning task->prio必须 < 256                              
      10797 + * @warning 必须持有rq锁                                       
       

      10798 + */                                                         
      10799 +static void fifo_sched_enqueue(struct rq *rq, TCB_t *task)  
            +{                                                           
      10800 +    struct fifo_rq *fifo_rq;                                
      10801 +    struct list_head *queue;                                
      10802 +                                                            
      10803 +    /* 参数验证 */                                             
      

      10804 +    if ((rq == NULL) || (task == NULL)) {                   
      10805 +        return;                                             
      10806 +    }                                                       
      10807 +                                                            
      10808 +    fifo_rq = rq->fifo_rq;                                  
      10809 +    if (fifo_rq == NULL) {                                  
      10810 +        return;                                             
      10811 +    }                                                       
      10812 +                                                            
      10813 +    /* 获取优先级队列 */                                       
         

      10814 +    if (task->prio >= 256U) {                               
      10815 +        return;  /* 无效优先级 */                              
       

      10816 +    }                                                       
      10817 +                                                            
      10818 +    queue = &fifo_rq->queue_array[task->prio];              
      10819 +                                                            
      10820 +    /* 加入队列尾部 */                                         
        

      10821 +    list_add_tail(&task->run_list, queue);                  
      10822 +                                                            
      10823 +    /* 更新位图 */                                             
      

      10824 +    set_bit(task->prio, fifo_rq->priority_bitmap);          
      10825 +}                                                           
      10826 +```                                                         
      10827 +                                                            
      10828 +#### 26.3.3 FIFO选择下一个任务                                 
         

      10829 +                                                            
      10830 +```c                                                        
      10831 +/**                                                         
      10832 + * @brief FIFO选择下一个任务                                   
         

      10833 + * @param rq 运行队列指针                                      
        

      10834 + * @return 任务控制块指针，无任务返回NULL                      
               

      10835 + *                                                          
      10836 + * @details 选择最高优先级任务                                 
           

      10837 + *                                                          
      10838 + * @note MISRA规则遵守：                                       
       

      10839 + *   - 规则13.5: 检查指针参数                                  
          

      10840 + *   - 规则21.3: 数组索引范围                                  
          

      10841 + *   - 规则18.4: 指针差值不超过数组大小                        
               

      10842 + *   - 规则1.3: signedness一致性                               
       

      10843 + *                                                          
      10844 + * @note 时间复杂度: O(1)                                      
       

      10845 + * @note 使用CLZ指令快速查找                                   
          

      10846 + * @note 开销: ~50ns                                          
      10847 + */                                                         
      10848 +static TCB_t *fifo_sched_pick_next(struct rq *rq) {         
      10849 +    struct fifo_rq *fifo_rq;                                
      10850 +    struct list_head *queue;                                
      10851 +    TCB_t *task;                                            
      10852 +    uint32_t priority;                                      
      10853 +    uint64_t bitmap_u64;                                    
      10854 +    uint32_t word_idx;                                      
      10855 +    uint32_t bit_offset;                                    
      10856 +                                                            
      10857 +    /* 参数验证 */                                             
      

      10858 +    if (rq == NULL) {                                       
      10859 +        return NULL;                                        
      10860 +    }                                                       
      10861 +                                                            
      10862 +    fifo_rq = rq->fifo_rq;                                  
      10863 +    if (fifo_rq == NULL) {                                  
      10864 +        return NULL;                                        
      10865 +    }                                                       
      10866 +                                                            
      10867 +    /* 查找最高优先级（使用CLZ） */                            
             

      10868 +    priority = 255U;                                        
      10869 +                                                            
      10870 +    /* 遍历4个64位字（256位） */                               
          

      10871 +    for (word_idx = 0U; word_idx < 4U; word_idx++) {        
      10872 +        bitmap_u64 = fifo_rq->priority_bitmap[word_idx];    
      10873 +                                                            
      10874 +        /* 跳过空字 */                                         
      

      10875 +        if (bitmap_u64 == 0ULL) {                           
      10876 +            continue;                                       
      10877 +        }                                                   
      10878 +                                                            
      10879 +        /* 使用CLZ查找最高位 */                                
         

      10880 +        bit_offset = (uint32_t)__builtin_clzll(bitmap_u64); 
      10881 +        priority = (word_idx * 64U) + (63U - bit_offset);   
      10882 +        break;                                              
      10883 +    }                                                       
      10884 +                                                            
      10885 +    /* 检查是否找到有效优先级 */                               
             

      10886 +    if (priority == 255U) {                                 
      10887 +        return NULL;  /* 队列为空 */                           
      

      10888 +    }                                                       
      10889 +                                                            
      10890 +    /* 获取优先级队列 */                                       
         

      10891 +    if (priority >= 256U) {                                 
      10892 +        return NULL;  /* 防御性编程 */                         
       

      10893 +    }                                                       
      10894 +                                                            
      10895 +    queue = &fifo_rq->queue_array[priority];                
      10896 +                                                            
      10897 +    /* 检查队列是否为空 */                                     
          

      10898 +    if (list_empty(queue) != 0) {                           
      10899 +        return NULL;                                        
      10900 +    }                                                       
      10901 +                                                            
      10902 +    /* 获取第一个任务 */                                       
         

      10903 +    task = list_first_entry(queue, TCB_t, run_list);        
      10904 +                                                            
      10905 +    return task;                                            
      10906 +}                                                           
      10907 +```                                                         
      10908 +                                                            
      10909 +### 26.4 EDF调度类实现MISRA-C编码规范                          
           

      10910 +                                                            
      10911 +#### 26.4.1 EDF入队                                           
      10912 +                                                            
      10913 +```c                                                        
      10914 +/**                                                         
      10915 + * @brief EDF任务入队                                          
      

      10916 + * @param rq 运行队列指针                                      
        

      10917 + * @param task 任务控制块指针                                  
         

      10918 + *                                                          
      10919 + * @details 按截止时间排序，插入红黑树                         
               

      10920 + *                                                          
      10921 + * @note MISRA规则遵守：                                       
       

      10922 + *   - 规则13.5: 检查指针参数                                  
          

      10923 + *   - 规则18.1: 指针间接访问                                  
          

      10924 + *   - 规则17.3: 类型匹配                                      
        

      10925 + *   - 规则21.3: 边界检查                                      
        

      10926 + *                                                          
      10927 + * @note 时间复杂度: O(log n)                                  
       

      10928 + * @note 使用红黑树维护                                        
         

      10929 + * @note 截止时间最早的在前                                    
           

      10930 + *                                                          
      10931 + * @warning task->deadline必须初始化                           
       

      10932 + * @warning 必须持有rq锁                                       
       

      10933 + */                                                         
      10934 +static void edf_sched_enqueue(struct rq *rq, TCB_t *task) { 
      10935 +    struct edf_rq *edf_rq;                                  
      10936 +    struct rb_node **link;                                  
      10937 +    struct rb_node *parent;                                 
      10938 +    TCB_t *entry;                                           
      10939 +                                                            
      10940 +    /* 参数验证 */                                             
      

      10941 +    if ((rq == NULL) || (task == NULL)) {                   
      10942 +        return;                                             
      10943 +    }                                                       
      10944 +                                                            
      10945 +    edf_rq = rq->edf_rq;                                    
      10946 +    if (edf_rq == NULL) {                                   
      10947 +        return;                                             
      10948 +    }                                                       
      10949 +                                                            
      10950 +    /* 检查截止时间有效性 */                                   
           

      10951 +    if (task->deadline == 0U) {                             
      10952 +        return;                                             
      10953 +    }                                                       
      10954 +                                                            
      10955 +    /* 查找插入位置 */                                         
        

      10956 +    link = &edf_rq->tasks_timeline.rb_node;                 
      10957 +    parent = NULL;                                          
      10958 +                                                            
      10959 +    while (*link != NULL) {                                 
      10960 +        parent = *link;                                     
      10961 +        entry = rb_entry(parent, TCB_t, run_node);          
      10962 +                                                            
      10963 +        /* 比较截止时间 */                                     
        

      10964 +        if (task->deadline < entry->deadline) {             
      10965 +            link = &(*link)->rb_left;                       
      10966 +        } else {                                            
      10967 +            link = &(*link)->rb_right;                      
      10968 +        }                                                   
      10969 +    }                                                       
      10970 +                                                            
      10971 +    /* 插入红黑树 */                                           
       

      10972 +    rb_link_node(&task->run_node, parent, link);            
      10973 +    rb_insert_color(&task->run_node,                        
            +&edf_rq->tasks_timeline);                                   
      10974 +                                                            
      10975 +    /* 更新计数 */                                             
      

      10976 +    edf_rq->nr_running++;                                   
      10977 +}                                                           
      10978 +```                                                         
      10979 +                                                            
      10980 +#### 26.4.2 EDF选择下一个任务                                  
         

      10981 +                                                            
      10982 +```c                                                        
      10983 +/**                                                         
      10984 + * @brief EDF选择下一个任务                                    
         

      10985 + * @param rq 运行队列指针                                      
        

      10986 + * @return 任务控制块指针，无任务返回NULL                      
               

      10987 + *                                                          
      10988 + * @details 选择截止时间最早的任务                             
             

      10989 + *                                                          
      10990 + * @note MISRA规则遵守：                                       
       

      10991 + *   - 规则13.5: 检查指针参数                                  
          

      10992 + *   - 规则18.4: 指针差值不超过对象大小                        
               

      10993 + *   - 规则1.3: signedness一致性                               
       

      10994 + *                                                          
      10995 + * @note 时间复杂度: O(1)                                      
       

      10996 + * @note 取红黑树最左节点                                      
          

      10997 + * @note 开销: ~70ns                                          
      10998 + */                                                         
      10999 +static TCB_t *edf_sched_pick_next(struct rq *rq) {          
      11000 +    struct edf_rq *edf_rq;                                  
      11001 +    struct rb_node *leftmost;                               
      11002 +    TCB_t *task;                                            
      11003 +                                                            
      11004 +    /* 参数验证 */                                             
      

      11005 +    if (rq == NULL) {                                       
      11006 +        return NULL;                                        
      11007 +    }                                                       
      11008 +                                                            
      11009 +    edf_rq = rq->edf_rq;                                    
      11010 +    if (edf_rq == NULL) {                                   
      11011 +        return NULL;                                        
      11012 +    }                                                       
      11013 +                                                            
      11014 +    /* 检查是否有任务 */                                       
         

      11015 +    if (edf_rq->nr_running == 0U) {                         
      11016 +        return NULL;                                        
      11017 +    }                                                       
      11018 +                                                            
      11019 +    /* 获取最左节点（截止时间最早） */                         
                

      11020 +    leftmost = rb_first(&edf_rq->tasks_timeline);           
      11021 +    if (leftmost == NULL) {                                 
      11022 +        return NULL;                                        
      11023 +    }                                                       
      11024 +                                                            
      11025 +    task = rb_entry(leftmost, TCB_t, run_node);             
      11026 +                                                            
      11027 +    return task;                                            
      11028 +}                                                           
      11029 +```                                                         
      11030 +                                                            
      11031 +### 26.5 CFS调度类实现MISRA-C编码规范                          
           

      11032 +                                                            
      11033 +#### 26.5.1 CFS入队                                           
      11034 +                                                            
      11035 +```c                                                        
      11036 +/**                                                         
      11037 + * @brief CFS任务入队                                          
      

      11038 + * @param rq 运行队列指针                                      
        

      11039 + * @param task 任务控制块指针                                  
         

      11040 + *                                                          
      11041 + * @details 按vruntime排序，插入红黑树                         
           

      11042 + *

  2. 形式化验证（seL4）

  - 8700+ 行代码 100% 验证
  - 零缓冲区溢出/空指针/整数溢出
  - 建议: 分层验证（关键模块优先）
  - 工具: Frama-C → CBMC → Isabelle/HOL

  3. 内存隔离增强

  Zephyr: Memory Domains
  NuttX: Protected Build (两块架构)

  综合方案:
  ProtectionDomain（保护域）
  ├── 内核域（KERNEL）
  ├── 驱动域（DRIVER）
  ├── 关键应用域（APP_CRITICAL）
  └── 普通应用域（APP_NORMAL）

  4. LSM 风格安全钩子（Linux）

  - 可扩展安全框架
  - 模块化策略
  - 建议: 轻量化实现（8-10个核心钩子）

  5. 栈溢出保护（Zephyr）

  - 金丝雀值 + 边界模式 + MPU 保护页
  - 多层防御

  🏗️ 架构合理性

  1. 微内核化（seL4 + QNX）

  当前: 单内核（驱动/FS/协议栈在内核）
  目标: 微内核（仅调度/内存/IPC）
  性能目标: IPC <100ns

  2. 自适应分区（QNX）

  - 动态 CPU 时间分配
  - 混合关键性系统支持
  - 符合 ARINC 653

  🔌 扩展性与灵活性

  1. AISafe-eBPF（简化版 eBPF）

  - 内核安全字节码执行
  - 无需重启的扩展性
  - 验证器保证安全性

  2. 模块化驱动框架

  - 统一设备抽象
  - 设备树集成
  - 自动探测

  3. RCU（Read-Copy-Update）

  - 读者零开销
  - 适合读多写少场景
