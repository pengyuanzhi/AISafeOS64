### 多锁分片优化文件锁表

将全局文件锁表从单一大锁改为 8 个独立分片锁，显著减少多线程竞争。

**问题**：
1. 全局 `s_locks[128]` 无锁保护
2. 所有文件锁操作（`find_lock`、`alloc_lock`、`fs_flock`）裸访问数组
3. 多线程同时修改锁表导致竞争

**解决方案**：使用 8 个分片锁，每个挂载点对应一个分片

### 架构设计

#### 分片锁数据结构

```c
#define FS_LOCK_SHARDS_COUNT  8U

typedef struct {
    TicketLock_t lock;           // 分片锁
    fs_lock_entry_t *locks;      // 锁表数组（每分片 16 个）
    uint32_t lock_count;         // 锁计数
} fs_lock_shard_t;

static fs_lock_shard_t s_lock_shards[FS_LOCK_SHARDS_COUNT];
```

#### 分片选择策略

```c
static inline uint32_t select_lock_shard(uint32_t mount_id) {
    return mount_id & 7;  // 位运算 O(1)，比取模更快
}
```

#### 初始化流程

```c
static void fs_ops_init(void) {
    // 清零挂载点表
    // ...

    // 为每个分片初始化 TicketLock 和锁表
    for (uint32_t i = 0U; i < FS_LOCK_SHARDS_COUNT; i++) {
        ticket_lock_init(&s_lock_shards[i].lock);
        s_lock_shards[i].locks = &s_locks[i * (FS_MAX_LOCKS / FS_LOCK_SHARDS_COUNT)];
        s_lock_shards[i].lock_count = 0U;
        // ...
    }
}
```

### 关键实现

#### 1. find_lock_in_shard（分片内查找）

```c
static fs_lock_entry_t *find_lock_in_shard(
    fs_lock_shard_t *shard,
    uint32_t mount_id,
    uint32_t ino)
{
    uint32_t i;
    uint32_t base = shard->index * (FS_MAX_LOCKS / FS_LOCK_SHARDS_COUNT);

    for (i = 0U; i < (FS_MAX_LOCKS / FS_LOCK_SHARDS_COUNT); i++) {
        fs_lock_entry_t *lock = &s_locks[base + i];
        if (lock->in_use &&
            lock->mount_id == mount_id &&
            lock->ino == ino) {
            return lock;
        }
    }
    return NULL;
}
```

#### 2. alloc_lock_in_shard（分片内分配）

```c
static fs_lock_entry_t *alloc_lock_in_shard(fs_lock_shard_t *shard) {
    uint32_t i;
    uint32_t base = shard->index * (FS_MAX_LOCKS / FS_LOCK_SHARDS_COUNT);

    for (i = 0U; i < (FS_MAX_LOCKS / FS_LOCK_SHARDS_COUNT); i++) {
        fs_lock_entry_t *lock = &s_locks[base + i];
        if (!lock->in_use) {
            memset(lock, 0, sizeof(fs_lock_entry_t));
            return lock;
        }
    }
    return NULL;
}
```

#### 3. fs_flock（文件锁操作）

```c
int32_t fs_flock(uint32_t mount_id, uint32_t ino,
                  fs_lock_type_t lock_type, uint32_t owner_tid)
{
    fs_lock_entry_t *lock;
    fs_lock_shard_t *shard = get_lock_shard(mount_id);

    // 获取分片锁
    ticket_lock_acquire(&shard->lock);

    if (lock_type == FS_LOCK_UN) {
        // 解锁操作
        lock = find_lock_in_shard(shard, mount_id, ino);
        if (lock == NULL || lock->lock.owner_tid != owner_tid) {
            ticket_lock_release(&shard->lock);
            return -1;
        }

        lock->lock.lock_count--;
        if (lock->lock.lock_count == 0U) {
            lock->in_use = false;
        }

        ticket_lock_release(&shard->lock);
        return 0;
    }
    else if (lock_type == FS_LOCK_SH || lock_type == FS_LOCK_EX) {
        // 加锁操作
        lock = find_lock_in_shard(shard, mount_id, ino);
        if (lock != NULL) {
            // 已有锁，检查是否可以嵌套
            if (lock->lock.owner_tid == owner_tid) {
                if (lock->lock.lock_type != lock_type) {
                    ticket_lock_release(&shard->lock);
                    return -1;
                }
                lock->lock.lock_count++;
                ticket_lock_release(&shard->lock);
                return 0;
            }
            else {
                // 不同线程的锁冲突
                ticket_lock_release(&shard->lock);
                return -1;
            }
        }

        // 分配新锁
        lock = alloc_lock_in_shard(shard);
        if (lock == NULL) {
            ticket_lock_release(&shard->lock);
            return -1;
        }

        lock->mount_id = mount_id;
        lock->ino = ino;
        lock->lock.locked = true;
        lock->lock.lock_type = lock_type;
        lock->lock.owner_tid = owner_tid;
        lock->lock.lock_count = 1U;
        lock->in_use = true;

        ticket_lock_release(&shard->lock);
        return 0;
    }

    ticket_lock_release(&shard->lock);
    return -1;
}
```

#### 4. fs_unmount（卸载文件系统）

```c
int32_t fs_unmount(uint32_t mount_id) {
    fs_mount_t *mnt = get_mount(mount_id);
    if (mnt == NULL) {
        return -1;
    }

    // 只锁定目标分片
    fs_lock_shard_t *shard = get_lock_shard(mount_id);
    ticket_lock_acquire(&shard->lock);

    // 清理该分片的所有锁
    uint32_t base = shard->index * (FS_MAX_LOCKS / FS_LOCK_SHARDS_COUNT);
    for (uint32_t i = 0U; i < (FS_MAX_LOCKS / FS_LOCK_SHARDS_COUNT); i++) {
        fs_lock_entry_t *lock = &s_locks[base + i];
        if (lock->in_use && lock->mount_id == mount_id) {
            lock->in_use = false;
            shard->lock_count--;
        }
    }

    ticket_lock_release(&shard->lock);

    // ... 其他卸载操作
}
```

### 技术特点

1. **O(1) 分片选择**：使用位运算（`mount_id & 7`）选择分片
2. **独立锁保护**：每个分片有独立的 `TicketLock`
3. **细粒度锁**：每个分片只保护 16 个锁项
4. **安全释放**：所有路径都经过 `ticket_lock_release`
5. **MISRA C:2012 合规**：4 空格缩进，Allman 括号，中文注释
